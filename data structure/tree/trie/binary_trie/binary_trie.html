struct TrieNode {
    int child[2] = {-1, -1};
    int cnt = 0;
};
vector<TrieNode> trie(1);  // trie[0] = gốc
const int BIT = 8;      // số bit đang dùng

void insert(int x) {
    int u = 0;
    trie[u].cnt++;
    for (int i = BIT - 1; i >= 0; i--) {
        int b = (x >> i) & 1;
        if (trie[u].child[b] == -1) {
            trie[u].child[b] = trie.size();
            trie.push_back(TrieNode());
        }
        u = trie[u].child[b];
        trie[u].cnt++;
    }
}
int queryMaxXor(int x) {
    int u = 0, res = 0;
    for (int i = BIT - 1; i >= 0; i--) {
        int b    = (x >> i) & 1;
        int want = b ^ 1;       // muốn bit ngược lại
        int nxt  = trie[u].child[want];
        if (nxt != -1 && trie[nxt].cnt > 0) {
            res |= (1 << i);      // đạt được bit khác -> +1 vào XOR
            u = nxt;
        } else {
            u = trie[u].child[b];  // đành chấp nhận bit giống
        }
    }
    return res;
}
