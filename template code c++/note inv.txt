ll power(ll a, ll b, ll m) {
    ll res = 1;
    a %= m;
    while (b) {
        if (b & 1) res = res * a % m;
        a = a * a % m;
        b >>= 1;
    }
    return res;
}

void precompute(int n) {
    // 1. Tính giai thừa
    fac[0] = 1;
    for (int i = 1; i <= n; i++)
        fac[i] = fac[i - 1] * i % MOD1;

    // 2. Tính nghịch đảo giai thừa tại n bằng Fermat, rồi suy ngược O(n)
    invFac[n] = power(fac[n], MOD1 - 2, MOD1);
    for (int i = n; i >= 1; i--)
        invFac[i - 1] = invFac[i] * i % MOD1;

    // 3. Tính nghịch đảo từng số từ 1 đến n bằng công thức truy hồi O(n)
    inv[1] = 1;
    for (int i = 2; i <= n; i++)
        inv[i] = (MOD1 - (MOD1 / i) * inv[MOD1 % i] % MOD1) % MOD1;
}


vector<ll> batchInverse(vector<ll>& a) {
    int n = a.size();
    vector<ll> prefix(n), invA(n);

    // 1. Tính prefix product: prefix[i] = a[0]*a[1]*...*a[i] mod MOD1
    prefix[0] = a[0] % MOD1;
    for (int i = 1; i < n; i++)
        prefix[i] = prefix[i - 1] * a[i] % MOD1;

    // 2. Tính nghịch đảo của TOÀN BỘ tích, chỉ 1 lần gọi power()
    ll invAll = power(prefix[n - 1], MOD1 - 2, MOD1);

    // 3. Duyệt ngược để tách nghịch đảo từng phần tử
    for (int i = n - 1; i >= 0; i--) {
        if (i > 0) {
            invA[i] = invAll * prefix[i - 1] % MOD1;
            invAll = invAll * a[i] % MOD1;
        } else {
            invA[i] = invAll;
        }
    }

    return invA;
}

ll C(int n, int k) {
    if (k < 0 || k > n) return 0;
    return fac[n] * invFac[k] % MOD1 * invFac[n - k] % MOD1;
}


inline void cal_fac_mod_inv_fac(long long num, long long mod) { fac_mod_[0] = fac_mod_[1] = 1; for (long long i = 2; i <= num; i++) { fac_mod_[i] = fac_mod_[i - 1] * i % mod; } inv_fac[num] = inv_(fac_mod_[num], mod); for (long long i = num; i >= 1; i--) { inv_fac[i - 1] = inv_fac[i] * i % mod; } }
inline long long nCk(long long n, long long k, long long mod) { if (k < 0 || k > n) return 0; return fac[n] * inv_fac[k] % mod * inv_fac[n - k] % mod; }
